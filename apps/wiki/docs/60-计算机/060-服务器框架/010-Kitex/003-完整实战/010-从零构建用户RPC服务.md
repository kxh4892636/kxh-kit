## 如何实现业务服务？

```go title="internal/user/service.go"
package user

import (
	"context"
	"errors"
	"fmt"
	"strings"
	"time"
)

type Service struct{ repo Repository }

func NewService(repo Repository) (*Service, error) {
	if repo == nil {
		return nil, errors.New("repository is required")
	}
	return &Service{repo: repo}, nil
}

func (s *Service) Create(ctx context.Context, in CreateInput) (*User, error) {
	in.Email = strings.ToLower(strings.TrimSpace(in.Email))
	in.Name = strings.TrimSpace(in.Name)
	if in.Email == "" || !strings.Contains(in.Email, "@") || len(in.Email) > 254 || in.Name == "" || len([]rune(in.Name)) > 100 {
		return nil, ErrInvalidInput
	}
	entity := &User{Email: in.Email, Name: in.Name, CreatedAt: time.Now().UTC()}
	if err := s.repo.Create(ctx, entity); err != nil {
		return nil, fmt.Errorf("create user: %w", err)
	}
	return entity, nil
}

func (s *Service) Get(ctx context.Context, id int64) (*User, error) {
	if id <= 0 {
		return nil, ErrInvalidInput
	}
	entity, err := s.repo.FindByID(ctx, id)
	if err != nil {
		return nil, fmt.Errorf("get user %d: %w", id, err)
	}
	return entity, nil
}
```

- 输入: 检查字段长度与正数 ID；文章标题按 Unicode 字符计数，正文按字节限制；作者不可变，更新和删除先检查所有权;

## 如何用 GORM 实现存储？

```go title="internal/user/repository_gorm.go"
package user

import (
	"context"
	"errors"
	"fmt"
	"gorm.io/gorm"
	"time"
)

type userRow struct {
	ID        int64     `gorm:"primaryKey"`
	Email     string    `gorm:"size:254;not null;uniqueIndex"`
	Name      string    `gorm:"size:100;not null"`
	CreatedAt time.Time `gorm:"not null"`
}

type GormRepository struct{ db *gorm.DB }

func NewGormRepository(db *gorm.DB) (*GormRepository, error) {
	if db == nil {
		return nil, errors.New("database is required")
	}
	return &GormRepository{db: db}, nil
}

func (r *GormRepository) Create(ctx context.Context, entity *User) error {
	row := userRow{Email: entity.Email, Name: entity.Name, CreatedAt: entity.CreatedAt}
	if err := r.db.WithContext(ctx).Create(&row).Error; err != nil {
		if errors.Is(err, gorm.ErrDuplicatedKey) {
			return ErrEmailConflict
		}
		return fmt.Errorf("insert user: %w", err)
	}
	entity.ID = row.ID
	return nil
}

func (r *GormRepository) FindByID(ctx context.Context, id int64) (*User, error) {
	var row userRow
	err := r.db.WithContext(ctx).First(&row, id).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return nil, ErrNotFound
	}
	if err != nil {
		return nil, fmt.Errorf("select user %d: %w", id, err)
	}
	return toEntity(row), nil
}
func (userRow) TableName() string { return "users" }
func toEntity(row userRow) *User {
	return &User{ID: row.ID, Email: row.Email, Name: row.Name, CreatedAt: row.CreatedAt}
}
```

- 表名: 使用 TableName 对齐 SQL 的 users/articles，避免被推导为 user_rows/article_rows;
- 错误: First 无记录转为领域错误；用户唯一键通过 TranslateError 转为 ErrDuplicatedKey;

## 如何读取配置并连接 MySQL？

```go title="internal/platform/database/open.go"
package database

import (
	"context"
	"database/sql"
	"errors"
	"gorm.io/driver/mysql"
	"gorm.io/gorm"
	"os"
	"time"
)

func Open() (*gorm.DB, *sql.DB, error) {
	dsn := os.Getenv("USER_DATABASE_DSN")
	if dsn == "" {
		return nil, nil, errors.New("USER_DATABASE_DSN is required")
	}
	db, err := gorm.Open(mysql.Open(dsn), &gorm.Config{TranslateError: true})
	if err != nil {
		return nil, nil, err
	}
	pool, err := db.DB()
	if err != nil {
		return nil, nil, err
	}
	pool.SetMaxOpenConns(10)
	pool.SetMaxIdleConns(5)
	pool.SetConnMaxLifetime(3 * time.Minute)
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()
	if err = pool.PingContext(ctx); err != nil {
		pool.Close()
		return nil, nil, err
	}
	return db, pool, nil
}
```

- 配置: USER_DATABASE_DSN 必填，密码和完整 DSN 不进入普通日志;
- 池: max open=10、max idle=5 是本地示例值；Ping 有 3 秒预算，退出时关闭 pool;

## 如何在启动前执行迁移？

```go title="cmd/migrate/main.go"
package main

import (
	"context"
	"database/sql"
	"errors"
	"example.com/user-service/internal/platform/database"
	"log"
	"os"
	"time"
)

func run() error {
	_, pool, err := database.Open()
	if err != nil {
		return err
	}
	defer pool.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	conn, err := pool.Conn(ctx)
	if err != nil {
		return err
	}
	defer conn.Close()
	var locked int
	if err = conn.QueryRowContext(ctx, "SELECT GET_LOCK('notes_schema_migration', 10)").Scan(&locked); err != nil {
		return err
	}
	if locked != 1 {
		return errors.New("migration lock unavailable")
	}
	defer func() {
		release, cancel := context.WithTimeout(context.Background(), time.Second)
		defer cancel()
		conn.ExecContext(release, "SELECT RELEASE_LOCK('notes_schema_migration')")
	}()
	if _, err = conn.ExecContext(ctx, "CREATE TABLE IF NOT EXISTS schema_migrations (version BIGINT PRIMARY KEY)"); err != nil {
		return err
	}
	var version int
	err = conn.QueryRowContext(ctx, "SELECT version FROM schema_migrations WHERE version=1").Scan(&version)
	if err == nil {
		return nil
	}
	if !errors.Is(err, sql.ErrNoRows) {
		return err
	}
	ddl, err := os.ReadFile("migrations/001_create_users.sql")
	if err != nil {
		return err
	}
	if _, err = conn.ExecContext(ctx, string(ddl)); err != nil {
		return err
	}
	_, err = conn.ExecContext(ctx, "INSERT INTO schema_migrations(version) VALUES (1)")
	return err
}
func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}
```

- 执行位置: 从项目根目录运行，读取 migrations 下的 SQL；版本 1 成功后重跑会跳过;
- 并发: 命名锁与 DDL 使用同一连接，避免两个初始化进程同时建表;
- 中断恢复: MySQL DDL 不随普通事务回滚；若建表成功但版本记录失败，需要人工核对表结构再修复版本记录，不能宣称自动恢复;
- 后续开发: 使用团队统一的迁移工具，本命令只演示初始化;

## 如何把 RPC 请求交给业务服务？

```go title="internal/transport/kitex/handler.go"
package kitextransport

import (
	"context"
	"errors"
	userdomain "example.com/user-service/internal/user"
	"example.com/user-service/kitex_gen/user"
	"fmt"
	"github.com/cloudwego/kitex/pkg/kerrors"
)

type UserServiceImpl struct {
	service *userdomain.Service
}

func NewUserServiceImpl(service *userdomain.Service) (*UserServiceImpl, error) {
	if service == nil {
		return nil, errors.New("user service is required")
	}
	return &UserServiceImpl{service: service}, nil
}

func (h *UserServiceImpl) CreateUser(ctx context.Context, req *user.CreateUserRequest) (*user.User, error) {
	if req == nil {
		return nil, kerrors.NewBizStatusError(40001, "request is required")
	}
	entity, err := h.service.Create(ctx, userdomain.CreateInput{Email: req.Email, Name: req.Name})
	switch {
	case errors.Is(err, userdomain.ErrInvalidInput):
		return nil, kerrors.NewBizStatusError(40001, "invalid user")
	case errors.Is(err, userdomain.ErrEmailConflict):
		return nil, kerrors.NewBizStatusError(40901, "email already exists")
	case err != nil:
		return nil, fmt.Errorf("create user: %w", err)
	default:
		return toRPCUser(entity), nil
	}
}

func (h *UserServiceImpl) GetUser(ctx context.Context, req *user.GetUserRequest) (*user.User, error) {
	if req == nil {
		return nil, kerrors.NewBizStatusError(40001, "request is required")
	}
	entity, err := h.service.Get(ctx, req.Id)
	switch {
	case errors.Is(err, userdomain.ErrInvalidInput):
		return nil, kerrors.NewBizStatusError(40001, "invalid user id")
	case errors.Is(err, userdomain.ErrNotFound):
		return nil, kerrors.NewBizStatusError(40401, "user not found")
	case err != nil:
		return nil, fmt.Errorf("get user: %w", err)
	default:
		return toRPCUser(entity), nil
	}
}
func toRPCUser(u *userdomain.User) *user.User {
	return &user.User{Id: u.ID, Email: u.Email, Name: u.Name, CreatedAtUnix: u.CreatedAt.Unix()}
}
```

- 适配: 输入检查、Service 调用、输出转换分别完成，业务错误码保持稳定；完整错误仅在服务端记录;

## 如何组装并启动进程？

```go title="cmd/server/main.go"
package main

import (
	"example.com/user-service/internal/platform/database"
	kitextransport "example.com/user-service/internal/transport/kitex"
	userdomain "example.com/user-service/internal/user"
	"example.com/user-service/kitex_gen/user/userservice"
	"github.com/cloudwego/kitex/pkg/rpcinfo"
	"github.com/cloudwego/kitex/pkg/transmeta"
	"github.com/cloudwego/kitex/server"
	"log"
	"net"
	"os"
	"time"
)

func run() error {
	db, pool, err := database.Open()
	if err != nil {
		return err
	}
	defer pool.Close()
	repo, err := userdomain.NewGormRepository(db)
	if err != nil {
		return err
	}
	svc, err := userdomain.NewService(repo)
	if err != nil {
		return err
	}
	handler, err := kitextransport.NewUserServiceImpl(svc)
	if err != nil {
		return err
	}
	address := os.Getenv("USER_SERVICE_ADDRESS")
	if address == "" {
		address = ":8888"
	}
	addr, err := net.ResolveTCPAddr("tcp", address)
	if err != nil {
		return err
	}
	svr := userservice.NewServer(handler, server.WithServiceAddr(addr),
		server.WithServerBasicInfo(&rpcinfo.EndpointBasicInfo{ServiceName: "user-service"}),
		server.WithExitWaitTime(5*time.Second), server.WithEnableContextTimeout(true),
		server.WithMetaHandler(transmeta.ServerTTHeaderHandler))
	return svr.Run()
}
func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}
```

- 入口: cmd/server 初始化 DB、Repository、Service、Handler，再创建 Server；TTHeader 和服务端 context 超时显式启用;
- 退出: Run 返回后关闭 DB；实际部署按需补注册发现与 readiness;
- 边界: 本例使用本地直连，不要求先部署治理平台;

## 如何用 Client 创建并查询用户？

```go title="cmd/client/main.go"
package main

import (
	"context"
	"example.com/user-service/kitex_gen/user"
	"example.com/user-service/kitex_gen/user/userservice"
	"fmt"
	"github.com/cloudwego/kitex/client"
	"github.com/cloudwego/kitex/pkg/kerrors"
	"github.com/cloudwego/kitex/pkg/transmeta"
	"github.com/cloudwego/kitex/transport"
	"log"
	"time"
)

func main() {
	cli, err := userservice.NewClient(
		"user-service",
		client.WithHostPorts("127.0.0.1:8888"),
		client.WithConnectTimeout(300*time.Millisecond),
		client.WithRPCTimeout(time.Second),
		client.WithTransportProtocol(transport.TTHeader),
		client.WithMetaHandler(transmeta.ClientTTHeaderHandler),
	)
	if err != nil {
		log.Fatalf("create user client: %v", err)
	}
	ctx, cancel := context.WithTimeout(context.Background(), time.Second)
	created, err := cli.CreateUser(ctx, &user.CreateUserRequest{Email: fmt.Sprintf("alice-%d@example.com", time.Now().UnixNano()), Name: "Alice"})
	cancel()
	if err != nil {
		if bizErr, ok := kerrors.FromBizStatusError(err); ok {
			log.Fatalf("create user rejected: code=%d message=%s", bizErr.BizStatusCode(), bizErr.BizMessage())
		}
		log.Fatalf("create user: %v", err)
	}
	found, err := cli.GetUser(context.Background(), &user.GetUserRequest{Id: created.Id})
	if err != nil {
		log.Fatalf("get user: %v", err)
	}
	log.Printf("user=%d %s %s", found.Id, found.Email, found.Name)
}
```

- 传输: 双方 TTHeader 与 MetaHandler 配套，Client 才能识别 BizStatus;
- 运行: 每次生成不同邮箱，便于重复验证成功路径；业务 Client 在进程内复用;

## 如何验证业务异常经过真实 RPC 传回？

```go title="internal/transport/kitex/rpc_test.go"
package kitextransport

import (
	"context"
	"net"
	"testing"
	"time"

	domain "example.com/user-service/internal/user"
	"example.com/user-service/kitex_gen/user"
	"example.com/user-service/kitex_gen/user/userservice"
	"github.com/cloudwego/kitex/client"
	"github.com/cloudwego/kitex/pkg/kerrors"
	"github.com/cloudwego/kitex/pkg/transmeta"
	"github.com/cloudwego/kitex/server"
	"github.com/cloudwego/kitex/transport"
)

func TestBizStatusOverTTHeader(t *testing.T) {
	listener, err := net.Listen("tcp", "127.0.0.1:0")
	if err != nil {
		t.Fatal(err)
	}
	svc, _ := domain.NewService(missingRepo{})
	h, _ := NewUserServiceImpl(svc)
	exit := make(chan error, 1)
	svr := userservice.NewServer(h, server.WithListener(listener),
		server.WithMetaHandler(transmeta.ServerTTHeaderHandler),
		server.WithExitSignal(func() <-chan error { return exit }))
	done := make(chan error, 1)
	go func() { done <- svr.Run() }()
	defer func() {
		exit <- nil
		select {
		case err := <-done:
			if err != nil {
				t.Error(err)
			}
		case <-time.After(10 * time.Second):
			t.Error("server did not stop")
		}
	}()
	cli, err := userservice.NewClient("user-service",
		client.WithShortConnection(),
		client.WithHostPorts(listener.Addr().String()),
		client.WithRPCTimeout(time.Second),
		client.WithTransportProtocol(transport.TTHeader),
		client.WithMetaHandler(transmeta.ClientTTHeaderHandler))
	if err != nil {
		t.Fatal(err)
	}
	_, err = cli.GetUser(context.Background(), &user.GetUserRequest{Id: 1})
	biz, ok := kerrors.FromBizStatusError(err)
	if !ok || biz.BizStatusCode() != 40401 {
		t.Fatalf("business status lost over wire: %v", err)
	}
}
```

- 范围: 临时 TCP 端口加生成 Client 验证 TTHeader 业务码；存储是 fake，测试用短连接便于退出;

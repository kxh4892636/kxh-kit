## 如何实现业务服务？

```go title="internal/article/service.go"
package article

import (
	"context"
	"errors"
	"fmt"
	"strings"
)

type Repository interface {
	Create(ctx context.Context, a *Article) error
	FindByID(ctx context.Context, id int64) (*Article, error)
	List(ctx context.Context, offset, limit int) ([]Article, int64, error)
	Update(ctx context.Context, a *Article) error
	Delete(ctx context.Context, id int64) error
}

type Service struct{ repo Repository }

func NewService(repo Repository) (*Service, error) {
	if repo == nil {
		return nil, errors.New("repository is required")
	}
	return &Service{repo: repo}, nil
}

func (s *Service) Create(ctx context.Context, in CreateInput) (*Article, error) {
	in.Title = strings.TrimSpace(in.Title)
	if in.AuthorID <= 0 || in.Title == "" || len([]rune(in.Title)) > 100 || in.Content == "" || len(in.Content) > 60000 {
		return nil, ErrInvalidInput
	}
	a := &Article{AuthorID: in.AuthorID, Title: in.Title, Content: in.Content}
	if err := s.repo.Create(ctx, a); err != nil {
		return nil, fmt.Errorf("create article: %w", err)
	}
	return a, nil
}

func (s *Service) Delete(ctx context.Context, actorID, articleID int64) error {
	a, err := s.repo.FindByID(ctx, articleID)
	if err != nil {
		return err
	}
	if a.AuthorID != actorID {
		return ErrPermissionDenied
	}
	return s.repo.Delete(ctx, articleID)
}
func (s *Service) Get(ctx context.Context, id int64) (*Article, error) {
	if id <= 0 {
		return nil, ErrInvalidInput
	}
	return s.repo.FindByID(ctx, id)
}
func (s *Service) List(ctx context.Context, page, size int) ([]Article, int64, error) {
	if page < 1 || page > 1000000 || size < 1 || size > 100 {
		return nil, 0, ErrInvalidInput
	}
	return s.repo.List(ctx, (page-1)*size, size)
}
func (s *Service) Update(ctx context.Context, actor, id int64, title, content string) (*Article, error) {
	title = strings.TrimSpace(title)
	if actor <= 0 || id <= 0 || title == "" || len([]rune(title)) > 100 || content == "" || len(content) > 60000 {
		return nil, ErrInvalidInput
	}
	a, err := s.repo.FindByID(ctx, id)
	if err != nil {
		return nil, err
	}
	if a.AuthorID != actor {
		return nil, ErrPermissionDenied
	}
	a.Title = title
	a.Content = content
	if err = s.repo.Update(ctx, a); err != nil {
		return nil, fmt.Errorf("update article: %w", err)
	}
	return a, nil
}
```

- 输入: 检查字段长度与正数 ID；文章标题按 Unicode 字符计数，正文按字节限制；作者不可变，更新和删除先检查所有权;

## 如何用 GORM 实现存储？

```go title="internal/article/repository_gorm.go"
package article

import (
	"context"
	"errors"
	"fmt"
	"gorm.io/gorm"
	"time"
)

type articleRow struct {
	ID        int64  `gorm:"primaryKey"`
	AuthorID  int64  `gorm:"not null;index"`
	Title     string `gorm:"size:100;not null"`
	Content   string `gorm:"type:text;not null"`
	CreatedAt time.Time
	UpdatedAt time.Time
}

type GormRepository struct{ db *gorm.DB }

func NewGormRepository(db *gorm.DB) *GormRepository {
	return &GormRepository{db: db}
}

func (r *GormRepository) FindByID(ctx context.Context, id int64) (*Article, error) {
	var row articleRow
	err := r.db.WithContext(ctx).First(&row, id).Error
	if errors.Is(err, gorm.ErrRecordNotFound) {
		return nil, ErrNotFound
	}
	if err != nil {
		return nil, fmt.Errorf("find article %d: %w", id, err)
	}
	return &Article{ID: row.ID, AuthorID: row.AuthorID, Title: row.Title, Content: row.Content}, nil
}
func (articleRow) TableName() string { return "articles" }
func (r *GormRepository) Create(ctx context.Context, a *Article) error {
	row := articleRow{AuthorID: a.AuthorID, Title: a.Title, Content: a.Content}
	if err := r.db.WithContext(ctx).Create(&row).Error; err != nil {
		return err
	}
	a.ID = row.ID
	return nil
}
func (r *GormRepository) List(ctx context.Context, offset, limit int) ([]Article, int64, error) {
	q := r.db.WithContext(ctx).Model(&articleRow{})
	var total int64
	if err := q.Count(&total).Error; err != nil {
		return nil, 0, err
	}
	var rows []articleRow
	if err := q.Order("id DESC").Offset(offset).Limit(limit).Find(&rows).Error; err != nil {
		return nil, 0, err
	}
	items := make([]Article, 0, len(rows))
	for _, a := range rows {
		items = append(items, Article{ID: a.ID, AuthorID: a.AuthorID, Title: a.Title, Content: a.Content})
	}
	return items, total, nil
}
func (r *GormRepository) Update(ctx context.Context, a *Article) error {
	// Explicit fields preserve empty values; mysql clientFoundRows makes no-op PUT count as matched.
	tx := r.db.WithContext(ctx).Model(&articleRow{}).Where("id = ? AND author_id = ?", a.ID, a.AuthorID).
		Updates(map[string]any{"title": a.Title, "content": a.Content})
	if tx.Error != nil {
		return tx.Error
	}
	if tx.RowsAffected == 0 {
		return ErrNotFound
	}
	return nil
}
func (r *GormRepository) Delete(ctx context.Context, id int64) error {
	tx := r.db.WithContext(ctx).Delete(&articleRow{}, id)
	if tx.Error != nil {
		return tx.Error
	}
	if tx.RowsAffected == 0 {
		return ErrNotFound
	}
	return nil
}
```

- 表名: 使用 TableName 对齐 SQL 的 users/articles，避免被推导为 user_rows/article_rows;
- 错误: First 无记录转为领域错误；用户唯一键通过 TranslateError 转为 ErrDuplicatedKey;
- 更新: DSN 带 clientFoundRows=true，让相同内容的 PUT 按匹配行计数；本例并发编辑采用最后写入生效，列表与总数不保证同一快照;

## 如何读取配置并连接 MySQL？

```go title="internal/platform/database/open.go"
package database

import (
	"context"
	"database/sql"
	"errors"
	"gorm.io/driver/mysql"
	"gorm.io/gorm"
	"os"
	"time"
)

func Open() (*gorm.DB, *sql.DB, error) {
	dsn := os.Getenv("ARTICLE_DATABASE_DSN")
	if dsn == "" {
		return nil, nil, errors.New("ARTICLE_DATABASE_DSN is required")
	}
	db, err := gorm.Open(mysql.Open(dsn), &gorm.Config{TranslateError: true})
	if err != nil {
		return nil, nil, err
	}
	pool, err := db.DB()
	if err != nil {
		return nil, nil, err
	}
	pool.SetMaxOpenConns(10)
	pool.SetMaxIdleConns(5)
	pool.SetConnMaxLifetime(3 * time.Minute)
	ctx, cancel := context.WithTimeout(context.Background(), 3*time.Second)
	defer cancel()
	if err = pool.PingContext(ctx); err != nil {
		pool.Close()
		return nil, nil, err
	}
	return db, pool, nil
}
```

- 配置: ARTICLE_DATABASE_DSN 必填，密码和完整 DSN 不进入普通日志;
- 池: max open=10、max idle=5 是本地示例值；Ping 有 3 秒预算，退出时关闭 pool;

## 如何在启动前执行迁移？

```go title="cmd/migrate/main.go"
package main

import (
	"context"
	"database/sql"
	"errors"
	"example.com/article-api/internal/platform/database"
	"log"
	"os"
	"time"
)

func run() error {
	_, pool, err := database.Open()
	if err != nil {
		return err
	}
	defer pool.Close()
	ctx, cancel := context.WithTimeout(context.Background(), 30*time.Second)
	defer cancel()
	conn, err := pool.Conn(ctx)
	if err != nil {
		return err
	}
	defer conn.Close()
	var locked int
	if err = conn.QueryRowContext(ctx, "SELECT GET_LOCK('notes_schema_migration', 10)").Scan(&locked); err != nil {
		return err
	}
	if locked != 1 {
		return errors.New("migration lock unavailable")
	}
	defer func() {
		release, cancel := context.WithTimeout(context.Background(), time.Second)
		defer cancel()
		conn.ExecContext(release, "SELECT RELEASE_LOCK('notes_schema_migration')")
	}()
	if _, err = conn.ExecContext(ctx, "CREATE TABLE IF NOT EXISTS schema_migrations (version BIGINT PRIMARY KEY)"); err != nil {
		return err
	}
	var version int
	err = conn.QueryRowContext(ctx, "SELECT version FROM schema_migrations WHERE version=1").Scan(&version)
	if err == nil {
		return nil
	}
	if !errors.Is(err, sql.ErrNoRows) {
		return err
	}
	ddl, err := os.ReadFile("migrations/001_create_articles.sql")
	if err != nil {
		return err
	}
	if _, err = conn.ExecContext(ctx, string(ddl)); err != nil {
		return err
	}
	_, err = conn.ExecContext(ctx, "INSERT INTO schema_migrations(version) VALUES (1)")
	return err
}
func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}
```

- 执行位置: 从项目根目录运行，读取 migrations 下的 SQL；版本 1 成功后重跑会跳过;
- 并发: 命名锁与 DDL 使用同一连接，避免两个初始化进程同时建表;
- 中断恢复: MySQL DDL 不随普通事务回滚；若建表成功但版本记录失败，需要人工核对表结构再修复版本记录，不能宣称自动恢复;
- 后续开发: 使用团队统一的迁移工具，本命令只演示初始化;

## 如何验证写请求身份？

```go title="internal/platform/auth/auth.go"
package auth

import (
	"context"
	"errors"
	"github.com/cloudwego/hertz/pkg/app"
	"github.com/golang-jwt/jwt/v5"
	"strconv"
	"strings"
	"time"
)

func Sign(key []byte, actor int64) (string, error) {
	if len(key) < 32 || actor <= 0 {
		return "", errors.New("invalid signing input")
	}
	claims := jwt.RegisteredClaims{Subject: strconv.FormatInt(actor, 10), Issuer: "notes-local", Audience: jwt.ClaimStrings{"article-api"}, ExpiresAt: jwt.NewNumericDate(time.Now().Add(15 * time.Minute)), IssuedAt: jwt.NewNumericDate(time.Now())}
	return jwt.NewWithClaims(jwt.SigningMethodHS256, claims).SignedString(key)
}
func Verify(key []byte, raw string) (int64, error) {
	claims := new(jwt.RegisteredClaims)
	token, err := jwt.ParseWithClaims(raw, claims, func(t *jwt.Token) (any, error) { return key, nil }, jwt.WithValidMethods([]string{"HS256"}), jwt.WithIssuer("notes-local"), jwt.WithAudience("article-api"), jwt.WithExpirationRequired())
	if err != nil || !token.Valid {
		return 0, errors.New("invalid token")
	}
	id, err := strconv.ParseInt(claims.Subject, 10, 64)
	if err != nil || id <= 0 {
		return 0, errors.New("invalid subject")
	}
	return id, nil
}
func RequireWrites(key []byte) app.HandlerFunc {
	return func(ctx context.Context, c *app.RequestContext) {
		method := string(c.Method())
		if method == "GET" || method == "HEAD" || method == "OPTIONS" {
			c.Next(ctx)
			return
		}
		fields := strings.Fields(c.Request.Header.Get("Authorization"))
		var id int64
		var err error
		if len(fields) != 2 || !strings.EqualFold(fields[0], "Bearer") {
			err = errors.New("missing bearer token")
		} else {
			id, err = Verify(key, fields[1])
		}
		if err != nil {
			c.Header("WWW-Authenticate", "Bearer")
			c.AbortWithStatusJSON(401, map[string]any{"code": "UNAUTHENTICATED", "message": "authentication required", "request_id": c.GetString("request_id")})
			return
		}
		c.Set("user_id", id)
		c.Next(ctx)
	}
}
```

- 身份: GET/HEAD/OPTIONS 不要求 Token，其余请求验签后写入 user_id；Service 再检查文章作者;

## 如何把 HTTP 请求交给业务服务？

```go title="biz/handler/article/article_service.go"
package article

import (
	"context"
	"errors"
	model "example.com/article-api/biz/model/article"
	domain "example.com/article-api/internal/article"
	"github.com/cloudwego/hertz/pkg/app"
	"log"
)

// Inject runs before generated handlers; each Engine owns its Service.
func Inject(svc *domain.Service) app.HandlerFunc {
	return func(ctx context.Context, c *app.RequestContext) { c.Set("article_service", svc); c.Next(ctx) }
}
func service(c *app.RequestContext) *domain.Service {
	v, _ := c.Get("article_service")
	return v.(*domain.Service)
}
func actor(c *app.RequestContext) int64 { v, _ := c.Get("user_id"); id, _ := v.(int64); return id }
func output(a *domain.Article) *model.Article {
	return &model.Article{ID: a.ID, AuthorID: a.AuthorID, Title: a.Title, Content: a.Content}
}
func fail(c *app.RequestContext, err error) {
	status, code, message := 500, "INTERNAL", "internal server error"
	switch {
	case errors.Is(err, domain.ErrInvalidInput):
		status, code, message = 400, "INVALID_ARGUMENT", "invalid article"
	case errors.Is(err, domain.ErrNotFound):
		status, code, message = 404, "ARTICLE_NOT_FOUND", "article not found"
	case errors.Is(err, domain.ErrPermissionDenied):
		status, code, message = 403, "PERMISSION_DENIED", "permission denied"
	default:
		log.Printf("request_id=%s error=%v", c.GetString("request_id"), err)
	}
	c.JSON(status, map[string]any{"code": code, "message": message, "request_id": c.GetString("request_id")})
}
func Create(ctx context.Context, c *app.RequestContext) {
	var req model.CreateArticleRequest
	if err := c.BindAndValidate(&req); err != nil {
		fail(c, domain.ErrInvalidInput)
		return
	}
	a, err := service(c).Create(ctx, domain.CreateInput{AuthorID: actor(c), Title: req.Title, Content: req.Content})
	if err != nil {
		fail(c, err)
		return
	}
	c.JSON(201, output(a))
}
func Get(ctx context.Context, c *app.RequestContext) {
	var req model.GetArticleRequest
	if err := c.BindAndValidate(&req); err != nil {
		fail(c, domain.ErrInvalidInput)
		return
	}
	a, err := service(c).Get(ctx, req.ID)
	if err != nil {
		fail(c, err)
		return
	}
	c.JSON(200, output(a))
}
func List(ctx context.Context, c *app.RequestContext) {
	req := model.NewListArticlesRequest()
	if err := c.BindAndValidate(req); err != nil {
		fail(c, domain.ErrInvalidInput)
		return
	}
	page, size := int(req.GetPage()), int(req.GetPageSize())
	items, total, err := service(c).List(ctx, page, size)
	if err != nil {
		fail(c, err)
		return
	}
	models := make([]*model.Article, 0, len(items))
	for i := range items {
		models = append(models, output(&items[i]))
	}
	c.JSON(200, &model.ListArticlesResponse{Items: models, Total: total, Page: int32(page), PageSize: int32(size)})
}
func Update(ctx context.Context, c *app.RequestContext) {
	var req model.UpdateArticleRequest
	if err := c.BindAndValidate(&req); err != nil {
		fail(c, domain.ErrInvalidInput)
		return
	}
	a, err := service(c).Update(ctx, actor(c), req.ID, req.Title, req.Content)
	if err != nil {
		fail(c, err)
		return
	}
	c.JSON(200, output(a))
}
func Delete(ctx context.Context, c *app.RequestContext) {
	var req model.GetArticleRequest
	if err := c.BindAndValidate(&req); err != nil || req.ID <= 0 {
		fail(c, domain.ErrInvalidInput)
		return
	}
	if err := service(c).Delete(ctx, actor(c), req.ID); err != nil {
		fail(c, err)
		return
	}
	c.Status(204)
}
```

- 适配: 每个 Engine 通过中间件注入 Service，避免包级单例；创建 201、读取或更新 200、删除 204，错误带 request_id;

```go title="main.go"
	"log"
	"os"
	"os/signal"
	"sync/atomic"
	"syscall"
	"time"
)

func makeEngine(svc *article.Service, key []byte) *server.Hertz {
	h := server.Default(server.WithHostPorts(":8888"), server.WithReadTimeout(3*time.Second), server.WithWriteTimeout(5*time.Second), server.WithMaxRequestBodySize(1<<20), server.WithExitWaitTime(5*time.Second))
	h.Use(func(ctx context.Context, c *app.RequestContext) {
		var id [16]byte
		if _, err := rand.Read(id[:]); err != nil {
			c.AbortWithStatus(500)
			return
		}
		requestID := hex.EncodeToString(id[:])
		c.Set("request_id", requestID)
		c.Header("X-Request-ID", requestID)
		ctx, cancel := context.WithTimeout(ctx, 2*time.Second)
		defer cancel()
		start := time.Now()
		c.Next(ctx)
		log.Printf("request_id=%s method=%s status=%d elapsed=%s", requestID, c.Method(), c.Response.StatusCode(), time.Since(start))
	})
	h.Use(auth.RequireWrites(key), handler.Inject(svc))
	register(h)
	return h
}
func run() error {
	key := []byte(os.Getenv("ARTICLE_JWT_KEY"))
	if len(key) < 32 {
		return errors.New("ARTICLE_JWT_KEY must contain at least 32 bytes")
	}
	db, pool, err := database.Open()
	if err != nil {
		return err
	}
	defer pool.Close()
	svc, err := article.NewService(article.NewGormRepository(db))
	if err != nil {
		return err
	}
	h := makeEngine(svc, key)
	var ready atomic.Bool
	ready.Store(true)
	h.GET("/livez", func(ctx context.Context, c *app.RequestContext) { c.Status(200) })
	h.GET("/readyz", func(ctx context.Context, c *app.RequestContext) {
		if !ready.Load() {
			c.Status(503)
			return
		}
		check, cancel := context.WithTimeout(ctx, 200*time.Millisecond)
		defer cancel()
		if pool.PingContext(check) != nil {
			c.Status(503)
			return
		}
		c.Status(200)
	})
	var startErr error
	h.SetCustomSignalWaiter(func(errs chan error) error {
		signals := make(chan os.Signal, 1)
		signal.Notify(signals, os.Interrupt, syscall.SIGTERM)
		defer signal.Stop(signals)
		select {
		case err := <-errs:
			startErr = err
			return err
		case <-signals:
			ready.Store(false)
			time.Sleep(time.Second)
			return nil
		}
	})
	h.Spin()
	return startErr
}
func main() {
	if err := run(); err != nil {
		log.Fatal(err)
	}
}
```

- 入口: 根 main.go 调用生成的 register(h)；Request ID、超时、认证、依赖注入依次安装;
- 退出: 先关闭 readiness，等待 1 秒示例摘流时间，再由 Spin 关闭，最后关闭 DB；按实际平台调整时间;
- 边界: Recovery 的默认 500 不保证业务错误 JSON 格式，需要统一时再配置自定义 Recovery;

```go title="main_test.go"
package main

import (
	"context"

type memoryRepo struct {
	rows map[int64]article.Article
	next int64
}

func (r *memoryRepo) Create(_ context.Context, a *article.Article) error {
	r.next++
	a.ID = r.next
	r.rows[a.ID] = *a
	return nil
}
func (r *memoryRepo) FindByID(_ context.Context, id int64) (*article.Article, error) {
	a, ok := r.rows[id]
	if !ok {
		return nil, article.ErrNotFound
	}
	return &a, nil
}
func (r *memoryRepo) List(_ context.Context, o, n int) ([]article.Article, int64, error) {
	items := []article.Article{}
	for i := r.next; i > 0; i-- {
		if a, ok := r.rows[i]; ok {
			items = append(items, a)
		}
	}
	total := len(items)
	if o > total {
		o = total
	}
	end := o + n
	if end > total {
		end = total
	}
	return items[o:end], int64(total), nil
}
func (r *memoryRepo) Update(_ context.Context, a *article.Article) error {
	r.rows[a.ID] = *a
	return nil
}
func (r *memoryRepo) Delete(_ context.Context, id int64) error { delete(r.rows, id); return nil }
func TestArticleLifecycle(t *testing.T) {
	repo := &memoryRepo{rows: map[int64]article.Article{}}
	svc, _ := article.NewService(repo)
	key := []byte("test-only-key-with-at-least-32-bytes")
	one, _ := auth.Sign(key, 1)
	two, _ := auth.Sign(key, 2)
	h := makeEngine(svc, key)
	request := func(method, url, body, token string, want int) []byte {
		t.Helper()
		headers := []ut.Header{{Key: "Content-Type", Value: "application/json"}}
		if token != "" {
			headers = append(headers, ut.Header{Key: "Authorization", Value: "Bearer " + token})
		}
		resp := ut.PerformRequest(h.Engine, method, url, &ut.Body{Body: strings.NewReader(body), Len: len(body)}, headers...).Result()
		if resp.StatusCode() != want {
			t.Fatalf("%s %s: want %d got %d: %s", method, url, want, resp.StatusCode(), resp.Body())
		}
		if resp.Header.Get("X-Request-ID") == "" {
			t.Fatal("missing request ID")
		}
		return append([]byte(nil), resp.Body()...)
	}
	body := request("POST", "/api/v1/articles", "{\"title\":\"Hertz\",\"content\":\"notes\"}", "", 401)
	if len(repo.rows) != 0 {
		t.Fatal("unauthenticated write")
	}
	body = request("POST", "/api/v1/articles", "{\"title\":\"Hertz\",\"content\":\"notes\"}", one, 201)
	var created struct{ ID int64 }
	if err := json.Unmarshal(body, &created); err != nil {
		t.Fatal(err)
	}
	url := fmt.Sprintf("/api/v1/articles/%d", created.ID)
	request("GET", url, "", "", 200)
	request("GET", "/api/v1/articles?page=1&page_size=20", "", "", 200)
	request("PUT", url, "{\"title\":\"Updated\",\"content\":\"notes\"}", two, 403)
	if repo.rows[created.ID].Title != "Hertz" {
		t.Fatal("forbidden update changed data")
	}
	request("PUT", url, "{\"title\":\"Updated\",\"content\":\"notes\"}", one, 200)
	request("POST", "/api/v1/articles", "{}", one, 400)
	request("GET", "/api/v1/articles?page_size=101", "", "", 400)
	request("DELETE", url, "", two, 403)
	if len(repo.rows) != 1 {
		t.Fatal("forbidden delete changed data")
	}
	request("DELETE", url, "", one, 204)
	request("GET", url, "", "", 404)
}
```
